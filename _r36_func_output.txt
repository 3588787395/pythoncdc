def get_trading_schedule(trading_time, is_backtest=True):
    time_set = set()
    start_time_delta = int(is_backtest)
    for s, e in trading_time:
        if s > 1200:
            continue
        for i in range(s + start_time_delta, e + 1):
            time_set.add(divmod(i, 60))
    else:
        return time_set
