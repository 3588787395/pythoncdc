```python
    def _r51m2_break_target_is_own_region_entry(
            self, block: BasicBlock, candidate_regions: List['Region']) -> bool:
        if block is None or not block.instructions:
            return False
        _is_own_region_entry = False
        for _r in candidate_regions:
            if _r is None or _r.entry is not block:
                continue
            if isinstance(_r, (IfRegion, LoopRegion, TryExceptRegion)):
                _is_own_region_entry = True
                break
        if not _is_own_region_entry:
            return False
        _eff = self.effective_instructions.get(block.start_offset) or []
        if len(_eff) < 2:
            return False
        if _eff[-1].opname not in CONDITIONAL_JUMP_OPS:
            return False
        _body = _eff[:-1]
        if not any(_i.opname not in NOISE_OPS and _i.opname != 'POP_TOP'
                   for _i in _body):
            return False
        _depth = 0
        for _i in _body:
            _push, _pop = self._stack_effect(_i)
            _depth += _push - _pop
            if _depth <= 0:
                return False
        return _depth > 0
```
