import re
def test(d, content):
    result = []
    for k, v in d.items():
        if not k < 'm':
            pattern = '(?<![.\\w])\\s*' + re.escape(k) + '\\('
            matches = re.findall(pattern, content)
            if len(matches) > 0 and k not in result:
                result.append(k)
            continue
        continue
    else:
        return result

    api_list = []
    for change_version, change_list in {'a': [1], 'b': [2]}.items():
        if not a < change_version <= b:
            pass
        for api_name in change_list:
            if api_name not in api_list:
                api_list.append(api_name)
            continue
    else:
        return api_list
