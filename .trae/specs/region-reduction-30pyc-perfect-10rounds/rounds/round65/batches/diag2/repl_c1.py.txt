                joined_str = {
                    'type': 'JoinedStr',
                    'values': fstring_parts,
                }
                # [R65-diag2 C1] 单三元 f-string 区域：消费点判定 + 后续语句拼接
                # 识别条件（同层次结构身份，纯指令/块语义；无函数名/文件名/偏移
                #   阈值/字面量计数判据）：本 TernaryRegion 的 merge_block 内含
                #   BUILD_STRING（JoinedStr 在本块内被组装），取**最后**一条
                #   BUILD_STRING 为组装点；其后第一条非 PRECALL/KW_NAMES 指令即
                #   该字符串的消费者，按操作码分派：RETURN_VALUE/RETURN_CONST =
                #   返回值、CALL(实参数==1) = 实参、STORE_* = 赋值目标。消费者不
                #   在这三者内、或 merge_block 无 BUILD_STRING（组装点不唯一，如
                #   `return f'{a if c else b}'`）时整体退回既有整块扫描行为，不
                #   新增逃逸口。
                # 归约方式：消费点把 merge_block 一分为二——「组装 + 消费」归约为
                #   本区域的单条语句；消费点之后**仍在同一线性区间内**的指令依
                #   「每块唯一归属」交 region.post_consumer_extra_stmts 归父语句
                #   序列（与既有 _try_wrap_fstring_pending_call /
                #   _try_build_ternary_merge_consumer_expr 同一约定，剥离隐式
                #   return None）。线性区间 = 消费点起，到 merge_block 内第一条
                #   「其它 CFG 块的 start_offset」（跳转目标 = 控制流汇点/嵌套结构
                #   入口，其指令已由对应结构区域归约，越界拼接会重复发射）或第一
                #   条跳转类指令为止；若该切点是条件跳转，其被测试的条件表达式前缀
                #   也一并留在区间外（它归属后继分支结构区域）。本分支原先只看
                #   「块内是否存在 RETURN_VALUE」来决定 return，使 f-string 之后的
                #   线性语句随块被标记 generated 而静默丢失（get_price 实测
                #   UNDER 42 中的 22 条）。
                # AST 映射：Return(JoinedStr) / Expr(Call(callee,[JoinedStr])) /
                #   Assign([Name], JoinedStr)，其后拼接线性区间指令归约出的语句
                #   序列。
                _r65d2_mb_eff = []
                if region.merge_block:
                    _r65d2_mb_eff = [i for i in region.merge_block.instructions
                                     if i.opname not in ('RESUME', 'NOP',
                                                         'CACHE', 'PUSH_NULL')]
                _r65d2_bs_pos = None
                for _r65d2_bi, _r65d2_ii in enumerate(_r65d2_mb_eff):
                    if _r65d2_ii.opname == 'BUILD_STRING':
                        _r65d2_bs_pos = _r65d2_bi
                _r65d2_stmt = None
                _r65d2_tail_from = None
                if _r65d2_bs_pos is not None:
                    try:
                        region.post_consumer_extra_stmts = None
                    except Exception:
                        pass
                    _r65d2_cj = _r65d2_bs_pos + 1
                    while (_r65d2_cj < len(_r65d2_mb_eff)
                           and _r65d2_mb_eff[_r65d2_cj].opname in (
                               'PRECALL', 'KW_NAMES')):
                        _r65d2_cj += 1
                    _r65d2_cons = (_r65d2_mb_eff[_r65d2_cj]
                                   if _r65d2_cj < len(_r65d2_mb_eff)
                                   else None)
                    if (_r65d2_cons is not None
                            and _r65d2_cons.opname in ('RETURN_VALUE',
                                                       'RETURN_CONST')):
                        _r65d2_stmt = {'type': 'Return', 'value': joined_str}
                    elif (_r65d2_cons is not None
                            and _r65d2_cons.opname == 'CALL'
                            and (_r65d2_cons.arg or 0) == 1):
                        _r65d2_stmt = self._try_wrap_fstring_pending_call(
                            region, region.merge_block, joined_str)
                        _r65d2_tail_from = _r65d2_cj + 1
                        if (_r65d2_tail_from < len(_r65d2_mb_eff)
                                and _r65d2_mb_eff[_r65d2_tail_from].opname
                                == 'POP_TOP'):
                            _r65d2_tail_from += 1
                    elif (_r65d2_cons is not None
                            and _r65d2_cons.opname in (
                                'STORE_FAST', 'STORE_NAME', 'STORE_GLOBAL',
                                'STORE_DEREF')):
                        _r65d2_stmt = {
                            'type': 'Assign',
                            'targets': [{'type': 'Name',
                                         'id': _r65d2_cons.argval,
                                         'ctx': 'Store'}],
                            'value': joined_str,
                        }
                        _r65d2_tail_from = _r65d2_cj + 1
                if _r65d2_stmt is not None:
                    _r65d2_ex = []
                    region.post_consumer_extra_stmts = None
                    # 幂等护栏：同一 TernaryRegion 在双角色入口场景会被驱动调用
                    # 两次（既有行为，非本次引入；landed 下同一条 Expr 也被发射
                    # 两次），且两次的返回值都会进入父语句序列。用区域对象上的一次
                    # 性标记保证线性区间语句只拼接一次。
                    # 双角色入口护栏（原则 2「每块唯一归属」，与既有 R59-B Fix1
                    # 同一判据）：merge 块同时是另一区域的 entry / merge_block 时，
                    # 线性区间语句的归属未定，本处不拼接，交由该区域自行发射。
                    _r65d2_shared = False
                    for _r65d2_o in self.regions:
                        if _r65d2_o is region:
                            continue
                        if (getattr(_r65d2_o, 'entry', None) is region.merge_block
                                or getattr(_r65d2_o, 'merge_block', None)
                                is region.merge_block):
                            _r65d2_shared = True
                            break
                    if (_r65d2_tail_from is not None
                            and not _r65d2_shared
                            and not getattr(region, '_r65d2_tail_done', False)):
                        region._r65d2_tail_done = True
                        _r65d2_blk_starts = set(
                            getattr(self.cfg, 'offset_to_block', {}) or {})
                        _r65d2_blk_starts.discard(
                            getattr(region.merge_block, 'start_offset', None))
                        _r65d2_tail = []
                        _r65d2_cut_i = len(_r65d2_mb_eff)
                        for _r65d2_k in range(_r65d2_tail_from,
                                              len(_r65d2_mb_eff)):
                            _r65d2_i = _r65d2_mb_eff[_r65d2_k]
                            if (_r65d2_i.offset in _r65d2_blk_starts
                                    or _r65d2_i.opname.startswith(
                                        ('JUMP_', 'POP_JUMP', 'FOR_ITER',
                                         'SEND'))):
                                _r65d2_cut_i = _r65d2_k
                                break
                        _r65d2_tail = list(
                            _r65d2_mb_eff[_r65d2_tail_from:_r65d2_cut_i])
                        # 切点是条件跳转时，被跳转测试的表达式不是线性语句而
                        # 是后继分支结构区域的判据（「每块唯一归属」：它归属该
                        # If/While 区域）。用与本分支 cond_val_start 同一栈效应
                        # 模型（_ternary_prefix_stack_effect）从切点回溯，把该
                        # 条件前缀一并排除在线性区间之外。
                        if (_r65d2_cut_i < len(_r65d2_mb_eff)
                                and _r65d2_mb_eff[_r65d2_cut_i].opname
                                    .startswith('POP_JUMP')):
                            _r65d2_needed = 1
                            for _r65d2_k in range(
                                    _r65d2_cut_i - 1,
                                    _r65d2_tail_from - 1, -1):
                                _r65d2_i = _r65d2_mb_eff[_r65d2_k]
                                if _r65d2_i.opname in (
                                        'STORE_FAST', 'STORE_NAME',
                                        'STORE_GLOBAL', 'STORE_DEREF'):
                                    _r65d2_tail = list(_r65d2_mb_eff[
                                        _r65d2_tail_from:_r65d2_k + 1])
                                    break
                                _r65d2_p, _r65d2_q = \
                                    self._ternary_prefix_stack_effect(_r65d2_i)
                                _r65d2_needed = _r65d2_needed - _r65d2_p + _r65d2_q
                                if _r65d2_needed <= 0:
                                    _r65d2_tail = list(_r65d2_mb_eff[
                                        _r65d2_tail_from:_r65d2_k])
                                    break
                        _r65d2_ex = []
                        if _r65d2_tail:
                            try:
                                _r65d2_ex = \
                                    self._build_statements_from_instructions(
                                        list(_r65d2_tail))
                            except Exception:
                                _r65d2_ex = []
                        while (_r65d2_ex
                               and isinstance(_r65d2_ex[-1], dict)
                               and _r65d2_ex[-1].get('type') == 'Return'
                               and isinstance(
                                   _r65d2_ex[-1].get('value'), dict)
                               and _r65d2_ex[-1]['value'].get('type')
                               == 'Constant'
                               and _r65d2_ex[-1]['value'].get('value')
                               is None):
                            _r65d2_ex = _r65d2_ex[:-1]
                        region.post_consumer_extra_stmts = (
                            _r65d2_ex or None)
                    results.append(_r65d2_stmt)
                    if _r65d2_ex:
                        self._emit_post_extra_with_if_upgrade(
                            results, _r65d2_ex, region)
                        region.post_consumer_extra_stmts = None
                else:
                    # 检查merge_block是否有RETURN_VALUE
                    has_return = False
                    if region.merge_block:
                        for instr in region.merge_block.instructions:
                            if instr.opname in ('RETURN_VALUE', 'RETURN_CONST'):
                                has_return = True
                                break
                    if has_return:
                        results.append({'type': 'Return', 'value': joined_str})
                    else:
                        results.append({'type': 'Expr', 'value': joined_str})
