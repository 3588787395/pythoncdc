SECTION 4: Root Cause Analysis
================================================================================

Mismatch function: system_func_with_execution_phase

Original source (inferred from bytecode):
  def system_func_with_execution_phase(execution_phase, func):
      from IQEngine.core import execution_context as exc_ctxt
      from IQEngine.const import ExcType
      def wrapper(*args, **kwargs):
          with exc_ctxt.ExecutionContext(execution_phase):
              with ModifyExceptionFromType(ExcType.SYSTEM_EXC):
                  return func(*(args), **(kwargs))
      return wrapper

Decompiled output (INCORRECT):
  def system_func_with_execution_phase(execution_phase, func):
      import IQEngine.core.execution_context
      def wrapper(*args, **kwargs):
          with exc_ctxt.ExecutionContext(execution_phase):
              with ModifyExceptionFromType(ExcType.SYSTEM_EXC):
                  return func(*(args), **(kwargs))
      return wrapper

Root cause: The decompiler (pycdc) incorrectly decompiled two import
statements that use "from X import Y" syntax and store the imported
names as closure variables (STORE_DEREF instead of STORE_FAST):

1. "from IQEngine.core import execution_context as exc_ctxt" was
   decompiled as "import IQEngine.core.execution_context". The
   original bytecode uses IMPORT_FROM+SWAP+POP_TOP+IMPORT_FROM+
   STORE_DEREF to extract "execution_context" and store it as
   "exc_ctxt" (a cell variable for the nested wrapper function).
   The decompiler lost the IMPORT_FROM and STORE_DEREF, instead
   generating a bare "import X.Y" which stores only the top-level
   package name as a local variable (STORE_FAST IQEngine).

2. "from IQEngine.const import ExcType" was completely lost in the
   decompiled output. The original bytecode has LOAD_CONST+
   IMPORT_NAME+IMPORT_FROM+STORE_DEREF+POP_TOP for this import,
   but none of these instructions appear in the recompiled code.

3. Because the two import-from statements were not properly
   decompiled, the closure variables (exc_ctxt, ExcType) that
   should be captured by the nested wrapper function are missing
   from the outer function. This causes the MAKE_FUNCTION closure
   tuple to be wrong: original has 4 closures (ExcType, exc_ctxt,
   execution_phase, func) but recompiled has only 2 (execution_phase,
   func). The wrapper function itself then references undefined
   names (exc_ctxt, ExcType) at runtime.

This is a known pycdc limitation: it fails to properly decompile
"from X import Y" statements when the imported name is stored as
a cell/free variable (STORE_DEREF) rather than a local variable
(STORE_FAST/STORE_NAME). This typically occurs when the imported
name is referenced by a nested function (closure).

def full_update(src_dict, drt_dict):
    """
    完全字典更新，支持所有类字典类型
    :param src_dict: collections.abc.Mapping 源字典
    :param drt_dict: collections.abc.Mapping 目标字典
    :return: None
    """
    for key, value in src_dict.items():
        if key in drt_dict and isinstance(drt_dict[key], collections.abc.Mapping) and isinstance(value, collections.abc.Mapping):
            full_update(value, drt_dict[key])
            continue
        drt_dict[key] = value
        continue
def get_original_function(func):
    """
    获取装饰器所装饰函数的原函数
    :param func: function 装饰器所装饰函数
    :return: function 原函数
    """
    func_wrapped = getattr(func, '__wrapped__', None)
    while func_wrapped:
        func = func_wrapped
        func_wrapped = getattr(func, '__wrapped__', None)
    return func

def value_parse(value):
    if value in ('True', 'true'):
        return True
    elif value in ('False', 'false'):
        return False
    elif value.isdigit():
        return int(value)
    try:
        return float(value)
        return None
    except ValueError:
        system_log.error(get_traceback_message())
    return value

def create_portfolio(engine):
    from IQEngine.account import Positions, Portfolio
    accounts = {}
    config = engine.config
    start_date = datetime.datetime.combine(config.strategy.start_date, datetime.time.min)
    units = 0
    for account_type, starting_cash in six.iteritems(config.strategy.accounts):
        if starting_cash == 0:
            error_message = _('{} 账户初始资金为0请设置').format(account_type)
            strategy_log.error(error_message)
            raise RuntimeError(error_message)
        account_model = engine.get_account_model(account_type)
        position_model = engine.get_position_model(account_type)
        positions = Positions(position_model)
        account = account_model(starting_cash, positions)
        units += account.total_value
        accounts[account_type] = account
    else:
        return Portfolio(config.strategy.start_date, 1, units, accounts)

def extend_to_enum(enum, extend_dict):
    """
    扩展枚举类项目
    :params enum: Enum 枚举类型
    :params extend_dict: dict 追加的项目和值的字典
    :return: None
    """
    if extend_dict.keys() & enum._member_map_.keys() or extend_dict.values() & enum._value2member_map_.keys():
        raise ValueError('extend_dict:{} is invalid'.format(extend_dict))
    for key, val in extend_dict.items():
        v = object.__new__(enum)
        v.__objclass__ = enum
        v._name_ = key
        v._value_ = val
        enum._member_map_[key] = v
        enum._member_names_.append(key)
        enum._value2member_map_[val] = v
def enum_clear(enum, extend_dict):
    """
    扩展枚举类项目
    :params enum: Enum 枚举类型
    :return: None
    """
    for key, val in extend_dict.items():
        del (enum._member_map_[key])
        enum._member_names_.remove(key)
        del (enum._value2member_map_[val])
